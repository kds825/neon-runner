[전역 실행 규칙]
- 이 파일의 Phase를 1번부터 6번까지 순서대로 빠짐없이 실행한다
- 각 Phase는 이전 Phase가 완전히 완료된 후에만 시작한다
- 절대로 사용자에게 확인을 요청하지 않는다
- 에러 발생 시: 오류 메시지 분석 → 원인 파악 → 수정 → 재실행 (최대 3회 자동 재시도)
- 3회 재시도 후에도 실패 시: 해당 기능을 최소 구현으로 대체하고 다음으로 진행
- 각 Phase 시작 시 "▶ Phase N 시작" 출력
- 각 Phase 완료 시 "✅ Phase N 완료" 출력 후 즉시 다음 Phase 진행
- 모든 터미널 명령은 직접 실행한다 (설명만 하고 멈추는 것 금지)
- 파일 수정 시 전체 파일 재작성 금지 — 변경 부분만 수정
- requestAnimationFrame 루프 내에서 new 키워드로 객체 생성 금지 (GC 방지)
- 모든 반복 생성 오브젝트는 Object Pool 패턴으로 관리

  import * as CANNON from 'cannon-es'

  export class Physics {
    public world: CANNON.World
    private fixedTimeStep = 1 / 60
    private maxSubSteps = 3

    constructor() {
      this.world = new CANNON.World({
        gravity: new CANNON.Vec3(0, -9.82, 0),
      })
      this.world.broadphase = new CANNON.SAPBroadphase(this.world)
      this.world.allowSleep = true

      // 기본 지면 contact material
      const groundMaterial = new CANNON.Material('ground')
      const wheelMaterial = new CANNON.Material('wheel')
      const contactMaterial = new CANNON.ContactMaterial(groundMaterial, wheelMaterial, {
        friction: 0.4,
        restitution: 0.1,
      })
      this.world.addContactMaterial(contactMaterial)
    }

    update(deltaTime: number) {
      this.world.step(this.fixedTimeStep, deltaTime, this.maxSubSteps)
    }
  }

--- 2-2. src/core/InputManager.ts 작성 ---

파일 경로: ./src/core/InputManager.ts

  export interface InputState {
    forward: boolean
    backward: boolean
    left: boolean
    right: boolean
    nitro: boolean
    brake: boolean
  }

  export class InputManager {
    private keys: Record<string, boolean> = {}

    constructor() {
      window.addEventListener('keydown', (e) => {
        this.keys[e.code] = true
        e.preventDefault()
      })
      window.addEventListener('keyup', (e) => {
        this.keys[e.code] = false
      })
    }

    get state(): InputState {
      return {
        forward:  this.keys['KeyW'] || this.keys['ArrowUp']    || false,
        backward: this.keys['KeyS'] || this.keys['ArrowDown']  || false,
        left:     this.keys['KeyA'] || this.keys['ArrowLeft']  || false,
        right:    this.keys['KeyD'] || this.keys['ArrowRight'] || false,
        nitro:    this.keys['Space'] || false,
        brake:    this.keys['ShiftLeft'] || this.keys['ShiftRight'] || false,
      }
    }
  }

--- 2-3. src/entities/PlayerCar.ts 작성 ---

파일 경로: ./src/entities/PlayerCar.ts

  import * as THREE from 'three'
  import * as CANNON from 'cannon-es'
  import type { InputState } from '../core/InputManager'

  export class PlayerCar {
    public mesh: THREE.Group
    public body: CANNON.Body
    public speed: number = 0
    public hp: number = 100

    private readonly MAX_SPEED = 60
    private readonly NITRO_SPEED = 90
    private readonly ENGINE_FORCE = 800
    private readonly BRAKE_FORCE = 400
    private readonly STEER_ANGLE = 0.6
    private readonly STEER_LERP = 0.12
    private currentSteer = 0

    // Nitro
    public nitroActive = false
    public nitroCooldown = false
    public nitroTimeLeft = 0
    public nitroCooldownLeft = 0
    private readonly NITRO_DURATION = 3
    private readonly NITRO_COOLDOWN = 10

    constructor(scene: THREE.Scene, world: CANNON.World) {
      // --- 물리 바디 ---
      const shape = new CANNON.Box(new CANNON.Vec3(1, 0.5, 2.2))
      this.body = new CANNON.Body({
        mass: 800,
        shape,
        linearDamping: 0.3,
        angularDamping: 0.99,
      })
      this.body.position.set(0, 1, 0)
      world.addBody(this.body)

      // --- 시각 메시 ---
      this.mesh = new THREE.Group()

      // 차체
      const bodyGeo = new THREE.BoxGeometry(2, 1, 4.4)
      const bodyMat = new THREE.MeshStandardMaterial({
        color: 0x00ffff,
        metalness: 0.8,
        roughness: 0.2,
        emissive: 0x003333,
      })
      const bodyMesh = new THREE.Mesh(bodyGeo, bodyMat)
      bodyMesh.castShadow = true
      this.mesh.add(bodyMesh)

      // 캐빈 (지붕)
      const cabinGeo = new THREE.BoxGeometry(1.6, 0.7, 2.2)
      const cabinMat = new THREE.MeshStandardMaterial({
        color: 0x001a2e,
        metalness: 0.9,
        roughness: 0.1,
      })
      const cabinMesh = new THREE.Mesh(cabinGeo, cabinMat)
      cabinMesh.position.set(0, 0.85, -0.3)
      this.mesh.add(cabinMesh)

      // 앞 헤드라이트
      const headlightGeo = new THREE.BoxGeometry(0.4, 0.15, 0.05)
      const headlightMat = new THREE.MeshStandardMaterial({
        color: 0xffffff,
        emissive: 0xffffff,
        emissiveIntensity: 2,
      })
      const leftLight = new THREE.Mesh(headlightGeo, headlightMat)
      leftLight.position.set(-0.6, 0.1, 2.2)
      this.mesh.add(leftLight)

      const rightLight = leftLight.clone()
      rightLight.position.set(0.6, 0.1, 2.2)
      this.mesh.add(rightLight)

      // 헤드라이트 PointLight
      const pointLight = new THREE.PointLight(0x00ffff, 3, 30)
      pointLight.position.set(0, 1, 3)
      this.mesh.add(pointLight)

      // 바퀴 (4개)
      const wheelPositions = [
        new THREE.Vector3(-1.1,  -0.5,  1.5),
        new THREE.Vector3( 1.1,  -0.5,  1.5),
        new THREE.Vector3(-1.1,  -0.5, -1.5),
        new THREE.Vector3( 1.1,  -0.5, -1.5),
      ]
      const wheelGeo = new THREE.CylinderGeometry(0.4, 0.4, 0.3, 16)
      const wheelMat = new THREE.MeshStandardMaterial({ color: 0x222222 })
      wheelPositions.forEach(pos => {
        const wheel = new THREE.Mesh(wheelGeo, wheelMat)
        wheel.rotation.z = Math.PI / 2
        wheel.position.copy(pos)
        wheel.castShadow = true
        this.mesh.add(wheel)
      })

      scene.add(this.mesh)
    }

    update(input: InputState, deltaTime: number) {
      // Nitro 타이머
      if (this.nitroActive) {
        this.nitroTimeLeft -= deltaTime
        if (this.nitroTimeLeft <= 0) {
          this.nitroActive = false
          this.nitroCooldown = true
          this.nitroCooldownLeft = this.NITRO_COOLDOWN
        }
      }
      if (this.nitroCooldown) {
        this.nitroCooldownLeft -= deltaTime
        if (this.nitroCooldownLeft <= 0) {
          this.nitroCooldown = false
        }
      }
      if (input.nitro && !this.nitroActive && !this.nitroCooldown) {
        this.nitroActive = true
        this.nitroTimeLeft = this.NITRO_DURATION
      }

      const maxSpd = this.nitroActive ? this.NITRO_SPEED : this.MAX_SPEED
      const forward = new CANNON.Vec3(
        -Math.sin(this.body.quaternion.toEuler(new CANNON.Vec3()).y),
        0,
        -Math.cos(this.body.quaternion.toEuler(new CANNON.Vec3()).y)
      )

      // 속도 계산
      const vel = this.body.velocity
      const spd = Math.sqrt(vel.x * vel.x + vel.z * vel.z)
      this.speed = spd

      // 엔진 힘
      if (input.forward) {
        const force = forward.scale(this.ENGINE_FORCE * (this.nitroActive ? 1.8 : 1))
        this.body.applyForce(force, this.body.position)
      }
      if (input.backward) {
        const force = forward.scale(-this.ENGINE_FORCE * 0.5)
        this.body.applyForce(force, this.body.position)
      }

      // 브레이크
      if (input.brake) {
        this.body.velocity.x *= (1 - this.BRAKE_FORCE * deltaTime / 1000)
        this.body.velocity.z *= (1 - this.BRAKE_FORCE * deltaTime / 1000)
      }

      // 최대 속도 클램핑
      if (spd > maxSpd) {
        this.body.velocity.x *= maxSpd / spd
        this.body.velocity.z *= maxSpd / spd
      }

      // 조향
      const targetSteer = input.left ? this.STEER_ANGLE : input.right ? -this.STEER_ANGLE : 0
      this.currentSteer += (targetSteer - this.currentSteer) * this.STEER_LERP

      if (spd > 1) {
        const turnFactor = this.currentSteer * (spd / maxSpd) * deltaTime * 1.5
        this.body.angularVelocity.y = this.currentSteer * Math.min(spd, 20) * 0.08
      }

      // 메시 동기화
      this.mesh.position.set(
        this.body.position.x,
        this.body.position.y,
        this.body.position.z
      )
      this.mesh.quaternion.set(
        this.body.quaternion.x,
        this.body.quaternion.y,
        this.body.quaternion.z,
        this.body.quaternion.w
      )
    }

    takeDamage(amount: number) {
      this.hp = Math.max(0, this.hp - amount)
    }

    get isDead(): boolean {
      return this.hp <= 0
    }
  }

목표: 운전석 시점으로 카메라를 고정하고, 속도에 따른 시각적 피드백을 구현한다.

--- 3-1. src/core/FirstPersonCamera.ts 작성 ---

파일 경로: ./src/core/FirstPersonCamera.ts

  import * as THREE from 'three'
  import type { PlayerCar } from '../entities/PlayerCar'

  export class FirstPersonCamera {
    private camera: THREE.PerspectiveCamera
    private baseFOV = 75
    private maxFOV = 100
    private currentFOV = 75
    private currentRoll = 0

    // 카메라 오프셋 (운전석 위치)
    private readonly OFFSET = new THREE.Vector3(0, 1.1, 0.3)
    private readonly LOOK_OFFSET = new THREE.Vector3(0, 0.9, 20) // 전방 응시점

    constructor(camera: THREE.PerspectiveCamera) {
      this.camera = camera
    }

    update(car: PlayerCar, steerAmount: number, deltaTime: number) {
      const carPos = car.mesh.position
      const carQuat = car.mesh.quaternion

      // 카메라 위치: 차량 로컬 오프셋을 월드 좌표로 변환
      const worldOffset = this.OFFSET.clone().applyQuaternion(carQuat)
      const camPos = carPos.clone().add(worldOffset)

      // 즉각 이동 (Lerp 없음 — 멀미 방지를 위해)
      this.camera.position.copy(camPos)

      // 전방 응시점
      const lookTarget = this.LOOK_OFFSET.clone().applyQuaternion(carQuat).add(carPos)
      this.camera.lookAt(lookTarget)

      // 조향 시 Roll 틸팅 (멀미 줄이기 위해 매우 미세하게)
      const targetRoll = -steerAmount * 0.03
      this.currentRoll += (targetRoll - this.currentRoll) * 0.08
      this.camera.rotation.z = this.currentRoll

      // 속도 비례 FOV 변화
      const speedRatio = Math.min(car.speed / 60, 1)
      const targetFOV = this.baseFOV + (this.maxFOV - this.baseFOV) * speedRatio
      this.currentFOV += (targetFOV - this.currentFOV) * 0.05
      this.camera.fov = this.currentFOV
      this.camera.updateProjectionMatrix()
    }
  }

    private readonly NUM_LANES = 4

    private readonly NEON_COLORS = [0x00ffff, 0xff00ff, 0x00ff88]

    constructor(scene: THREE.Scene, world: CANNON.World) {
      this.scene = scene
      this.world = world
      // 초기 3개 청크 생성
      for (let i = 0; i < 3; i++) {
        this.spawnChunk(-i * this.CHUNK_LENGTH)
      }
    }

    private spawnChunk(zStart: number) {
      const group = new THREE.Group()

      // 도로 바닥
      const roadGeo = new THREE.BoxGeometry(this.ROAD_WIDTH, 0.2, this.CHUNK_LENGTH)
      const roadMat = new THREE.MeshStandardMaterial({
        color: 0x0d0d1a,
        roughness: 0.9,
        metalness: 0.1,
      })
      const road = new THREE.Mesh(roadGeo, roadMat)
      road.receiveShadow = true
      group.add(road)

      // 차선 구분선 (흰색 점선)
      for (let lane = 1; lane < this.NUM_LANES; lane++) {
        const x = -this.ROAD_WIDTH / 2 + (lane / this.NUM_LANES) * this.ROAD_WIDTH
        for (let seg = 0; seg < 10; seg++) {
          const lineGeo = new THREE.BoxGeometry(0.15, 0.01, 8)
          const lineMat = new THREE.MeshStandardMaterial({
            color: 0xffffff,
            emissive: 0xaaaaaa,
            emissiveIntensity: 0.5,
          })
          const line = new THREE.Mesh(lineGeo, lineMat)
          line.position.set(x, 0.11, -this.CHUNK_LENGTH / 2 + seg * 20 + 5)
          group.add(line)
        }
      }

      // 도로 가드레일 (네온)
      const railGeo = new THREE.BoxGeometry(0.3, 1.2, this.CHUNK_LENGTH)
      const color = this.NEON_COLORS[Math.floor(Math.random() * this.NEON_COLORS.length)]
      const railMat = new THREE.MeshStandardMaterial({
        color,
        emissive: color,
        emissiveIntensity: 0.8,
        metalness: 0.9,
        roughness: 0.1,
      })
      const leftRail = new THREE.Mesh(railGeo, railMat)
      leftRail.position.set(-this.ROAD_WIDTH / 2, 0.5, 0)
      group.add(leftRail)

      const rightRail = leftRail.clone()
      rightRail.position.set(this.ROAD_WIDTH / 2, 0.5, 0)
      group.add(rightRail)

      // 가로등 (양쪽)
      for (let i = 0; i < 5; i++) {
        const lampZ = -this.CHUNK_LENGTH / 2 + i * 40 + 20
        const lampGeo = new THREE.CylinderGeometry(0.08, 0.08, 8)
        const lampMat = new THREE.MeshStandardMaterial({ color: 0x333333 })

        ;[-1, 1].forEach(side => {
          const lamp = new THREE.Mesh(lampGeo, lampMat)
          lamp.position.set(side * (this.ROAD_WIDTH / 2 + 1.5), 4, lampZ)
          group.add(lamp)

          const neonColor = this.NEON_COLORS[i % this.NEON_COLORS.length]
          const pointLight = new THREE.PointLight(neonColor, 2, 25)
          pointLight.position.set(side * (this.ROAD_WIDTH / 2 + 1), 7.5, lampZ)
          group.add(pointLight)
        })
      }

      // 배경 빌딩 (사이버펑크)
      for (let i = 0; i < 6; i++) {
        const h = 30 + Math.random() * 80
        const w = 8 + Math.random() * 12
        const bldGeo = new THREE.BoxGeometry(w, h, 15 + Math.random() * 20)
        const bldMat = new THREE.MeshStandardMaterial({
          color: 0x050510,
          emissive: this.NEON_COLORS[i % 3],
          emissiveIntensity: 0.05 + Math.random() * 0.1,
        })
        const bld = new THREE.Mesh(bldGeo, bldMat)
        const side = i % 2 === 0 ? -1 : 1
        bld.position.set(
          side * (this.ROAD_WIDTH / 2 + 15 + Math.random() * 30),
          h / 2,
          -this.CHUNK_LENGTH / 2 + Math.random() * this.CHUNK_LENGTH
        )
        group.add(bld)
      }

      group.position.set(0, 0, zStart - this.CHUNK_LENGTH / 2)
      this.scene.add(group)

      // 물리 지면
      const groundBody = new CANNON.Body({
        mass: 0,
        shape: new CANNON.Box(new CANNON.Vec3(this.ROAD_WIDTH / 2, 0.1, this.CHUNK_LENGTH / 2)),
      })
      groundBody.position.set(0, -0.1, zStart - this.CHUNK_LENGTH / 2)
      this.world.addBody(groundBody)

      this.chunks.push({ mesh: group, body: groundBody, zStart })
      this.totalGenerated++
    }

    update(playerZ: number) {
      // 플레이어 앞에 청크가 없으면 새로 생성
      const frontZ = playerZ - 300
      const lastChunk = this.chunks[this.chunks.length - 1]
      if (!lastChunk || lastChunk.zStart > frontZ) {
        this.spawnChunk(lastChunk ? lastChunk.zStart - this.CHUNK_LENGTH : 0)
      }

      // 뒤에 남겨진 청크 제거 (메모리 절약)
      while (this.chunks.length > 5) {
        const old = this.chunks.shift()!
        this.scene.remove(old.mesh)
        old.mesh.traverse(obj => {
          if (obj instanceof THREE.Mesh) {
            obj.geometry.dispose()
          }
        })
        this.world.removeBody(old.body)
      }
    }
  }

  import * as THREE from 'three'
  import * as CANNON from 'cannon-es'
  import { TrafficCar } from '../entities/TrafficCar'

  export class SpawnManager {
    private pool: TrafficCar[] = []
    private readonly POOL_SIZE = 20
    private readonly LANES = [-6, -2, 2, 6] // 4개 차선 X좌표
    private readonly SPAWN_DISTANCE = 250
    private readonly DESPAWN_DISTANCE = 50
    private spawnTimer = 0
    private spawnInterval = 2.5 // 초 (점점 줄어듦)

    constructor(scene: THREE.Scene, world: CANNON.World) {
      for (let i = 0; i < this.POOL_SIZE; i++) {
        this.pool.push(new TrafficCar(scene, world))
      }
    }

    update(playerZ: number, deltaTime: number, difficulty: number) {
      this.spawnTimer += deltaTime

      // 난이도에 따라 스폰 간격 감소
      this.spawnInterval = Math.max(0.8, 2.5 - difficulty * 0.3)

      if (this.spawnTimer >= this.spawnInterval) {
        this.spawnTimer = 0
        this.trySpawn(playerZ)
      }

      // 활성 차량 업데이트 및 범위 밖 차량 비활성화
      for (const car of this.pool) {
        if (car.active) {
          car.update()
          // 플레이어를 지나친 차량 비활성화
          if (car.body.position.z > playerZ + this.DESPAWN_DISTANCE) {
            car.deactivate()
          }
        }
      }
    }

    private trySpawn(playerZ: number) {
      const available = this.pool.find(c => !c.active)
      if (!available) return

      const lane = this.LANES[Math.floor(Math.random() * this.LANES.length)]
      const spawnZ = playerZ - this.SPAWN_DISTANCE
      available.spawn(lane, spawnZ)
    }

    getActiveCars(): TrafficCar[] {
      return this.pool.filter(c => c.active)
    }
  }

  export class ScoreSystem {
    public score = 0
    public distance = 0
    public combo = 1
    private comboTimer = 0
    private readonly COMBO_TIMEOUT = 3
    private readonly COMBO_MAX = 8
    private startZ = 0
    public bestScore = parseInt(localStorage.getItem('neonrunner_best') || '0')

    constructor(startZ: number) {
      this.startZ = startZ
    }

    update(playerZ: number, deltaTime: number) {
      this.distance = Math.max(0, this.startZ - playerZ) / 10

      // 거리 기반 점수
      this.score += this.distance * deltaTime * this.combo * 0.1

      // 콤보 타임아웃
      this.comboTimer += deltaTime
      if (this.comboTimer > this.COMBO_TIMEOUT) {
        this.resetCombo()
      }
    }

    addEvadeBonus() {
      this.combo = Math.min(this.combo + 1, this.COMBO_MAX)
      this.comboTimer = 0
      this.score += 100 * this.combo
    }

    resetCombo() {
      this.combo = 1
    }

    saveHighScore() {
      if (this.score > this.bestScore) {
        this.bestScore = Math.floor(this.score)
        localStorage.setItem('neonrunner_best', this.bestScore.toString())
      }
    }
  }

  const scene = new THREE.Scene()

  scene.fog = new THREE.FogExp2(0x03030f, 0.006)

  const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 800)
  const renderer = new THREE.WebGLRenderer({ antialias: true })
  renderer.setSize(window.innerWidth, window.innerHeight)
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2))
  renderer.shadowMap.enabled = true
  renderer.shadowMap.type = THREE.PCFSoftShadowMap
  renderer.toneMapping = THREE.ACESFilmicToneMapping
  renderer.toneMappingExposure = 1.5
  document.getElementById('canvas-container')!.appendChild(renderer.domElement)

  // Lights
  const ambient = new THREE.AmbientLight(0x0a0a2a, 3)
  scene.add(ambient)
  const dirLight = new THREE.DirectionalLight(0x8888ff, 2)
  dirLight.position.set(10, 30, 10)
  dirLight.castShadow = true
  scene.add(dirLight)

  // Stars (배경 별)
  const starsGeo = new THREE.BufferGeometry()
  const starCount = 3000
  const starPos = new Float32Array(starCount * 3)
  for (let i = 0; i < starCount * 3; i++) {
    starPos[i] = (Math.random() - 0.5) * 1000
  }
  starsGeo.setAttribute('position', new THREE.BufferAttribute(starPos, 3))
  const starsMat = new THREE.PointsMaterial({ color: 0xffffff, size: 0.3 })
  scene.add(new THREE.Points(starsGeo, starsMat))

  // --- Systems ---
  const physics = new Physics()
  const input = new InputManager()
  const car = new PlayerCar(scene, physics.world)
  const fpCamera = new FirstPersonCamera(camera)
  const road = new RoadGenerator(scene, physics.world)
  const spawner = new SpawnManager(scene, physics.world)
  const score = new ScoreSystem(car.body.position.z)
  const hud = new HUD()

  // Resize
  window.addEventListener('resize', () => {
    camera.aspect = window.innerWidth / window.innerHeight
    camera.updateProjectionMatrix()
    renderer.setSize(window.innerWidth, window.innerHeight)
  })

  // --- Collision Detection ---
  let lastDamageTime = 0
  function checkCollisions() {
    const carPos = car.body.position
    const activeCars = spawner.getActiveCars()
    const now = performance.now()

    for (const traffic of activeCars) {
      const dx = Math.abs(carPos.x - traffic.body.position.x)
      const dz = Math.abs(carPos.z - traffic.body.position.z)
      if (dx < 2.5 && dz < 4) {
        if (now - lastDamageTime > 800) {
          car.takeDamage(20)
          hud.showDamageFlash()
          score.resetCombo()
          lastDamageTime = now
        }
      } else if (dz > 6 && dz < 10 && traffic.active) {
        // 회피 성공 판정
        score.addEvadeBonus()
      }
    }
  }

  // --- Game Loop ---
  let lastTime = performance.now()
  let gameOver = false
  let difficulty = 0

  function animate(now: number) {
    if (gameOver) return
    requestAnimationFrame(animate)

    const delta = Math.min((now - lastTime) / 1000, 0.05)
    lastTime = now

    // 난이도 점진적 상승 (시간 기반)
    difficulty = Math.min(score.distance / 500, 5)

    // Physics
    physics.update(delta)

    // Systems
    car.update(input.state, delta)
    road.update(car.body.position.z)
    spawner.update(car.body.position.z, delta, difficulty)
    score.update(car.body.position.z, delta)

    // Camera
    fpCamera.update(car, car.currentSteer, delta)

    // Collision
    checkCollisions()

    // HUD
    hud.update(car, score)

    // Game Over 체크
    if (car.isDead) {
      gameOver = true
      score.saveHighScore()
      hud.showGameOver(score.score, score.bestScore)
      return
    }

    renderer.render(scene, camera)
  }
  requestAnimationFrame(animate)

      this.velocities = new Float32Array(this.COUNT * 3)
      this.lifetimes  = new Float32Array(this.COUNT)
      this.geo.setAttribute('position', new THREE.BufferAttribute(this.positions, 3))

      const mat = new THREE.PointsMaterial({
        color: 0xff6600,
        size: 0.5,
        transparent: true,
        depthWrite: false,
        blending: THREE.AdditiveBlending,
      })
      this.particles = new THREE.Points(this.geo, mat)
      this.particles.visible = false
      scene.add(this.particles)
    }

    explode(x: number, y: number, z: number) {
      this.active = true
      this.age = 0
      this.particles.visible = true
      for (let i = 0; i < this.COUNT; i++) {
        const i3 = i * 3
        this.positions[i3]     = x
        this.positions[i3 + 1] = y
        this.positions[i3 + 2] = z
        this.velocities[i3]     = (Math.random() - 0.5) * 12
        this.velocities[i3 + 1] = Math.random() * 8 + 2
        this.velocities[i3 + 2] = (Math.random() - 0.5) * 12
        this.lifetimes[i] = Math.random()
      }
    }

    update(delta: number) {
      if (!this.active) return
      this.age += delta
      const t = this.age / this.LIFE
      if (t >= 1) {
        this.active = false
        this.particles.visible = false
        return
      }
      ;(this.particles.material as THREE.PointsMaterial).opacity = 1 - t
      for (let i = 0; i < this.COUNT; i++) {
        const i3 = i * 3
        this.positions[i3]     += this.velocities[i3]     * delta
        this.positions[i3 + 1] += this.velocities[i3 + 1] * delta - 9.8 * delta * this.age
        this.positions[i3 + 2] += this.velocities[i3 + 2] * delta
      }
      this.geo.attributes.position.needsUpdate = true
    }
  }

  // 속도선 파티클 (빠를 때 사이드에 나타나는 선)
  export class SpeedLines {
    private particles: THREE.Points
    private geo: THREE.BufferGeometry
    private positions: Float32Array
    private readonly COUNT = 200

    constructor(scene: THREE.Scene) {
      this.geo = new THREE.BufferGeometry()
      this.positions = new Float32Array(this.COUNT * 3)
      this.geo.setAttribute('position', new THREE.BufferAttribute(this.positions, 3))
      const mat = new THREE.PointsMaterial({
        color: 0x00ffff,
        size: 0.15,
        transparent: true,
        opacity: 0,
        depthWrite: false,
        blending: THREE.AdditiveBlending,
      })
      this.particles = new THREE.Points(this.geo, mat)
      scene.add(this.particles)
    }

    update(carX: number, carY: number, carZ: number, speed: number) {
      const intensity = Math.max(0, (speed - 20) / 40)
      ;(this.particles.material as THREE.PointsMaterial).opacity = intensity * 0.6

      for (let i = 0; i < this.COUNT; i++) {
        const i3 = i * 3
        // 카메라 좌우에 랜덤 배치
        this.positions[i3]     = carX + (Math.random() - 0.5) * 14
        this.positions[i3 + 1] = carY + (Math.random() - 0.5) * 3
        this.positions[i3 + 2] = carZ - Math.random() * 30
      }
      this.geo.attributes.position.needsUpdate = true
    }
  }

  // 배기 연기 파티클
  export class ExhaustSmoke {
    private particles: THREE.Points
    private geo: THREE.BufferGeometry
    private positions: Float32Array
    private readonly COUNT = 30
    private index = 0

    constructor(scene: THREE.Scene) {
      this.geo = new THREE.BufferGeometry()
      this.positions = new Float32Array(this.COUNT * 3)
      this.geo.setAttribute('position', new THREE.BufferAttribute(this.positions, 3))
      const mat = new THREE.PointsMaterial({
        color: 0x334466,
        size: 0.8,
        transparent: true,
        opacity: 0.3,
        depthWrite: false,
      })
      this.particles = new THREE.Points(this.geo, mat)
      scene.add(this.particles)
    }

    emit(x: number, y: number, z: number) {
      const i3 = (this.index % this.COUNT) * 3
      this.positions[i3]     = x + (Math.random() - 0.5) * 0.5
      this.positions[i3 + 1] = y
      this.positions[i3 + 2] = z + Math.random() * 2
      this.index++
      this.geo.attributes.position.needsUpdate = true
    }
  }

  export class TouchControls {
    private joystickBase: HTMLElement
    private joystickKnob: HTMLElement
    private nitroBtn: HTMLElement
    private active = false
    private startX = 0
    private startY = 0
    private dx = 0
    private dy = 0

    public forward = false
    public backward = false
    public left = false
    public right = false
    public nitro = false

    constructor() {
      const isMobile = /Mobi|Android/i.test(navigator.userAgent)
        || window.innerWidth < 768

      if (!isMobile) return

      this.joystickBase = document.createElement('div')
      this.joystickBase.style.cssText = `
        position: fixed; bottom: 80px; left: 60px;
        width: 100px; height: 100px;
        background: rgba(0,255,255,0.1);
        border: 2px solid rgba(0,255,255,0.4);
        border-radius: 50%;
        z-index: 500; touch-action: none;
      `

      this.joystickKnob = document.createElement('div')
      this.joystickKnob.style.cssText = `
        position: absolute; top: 35px; left: 35px;
        width: 30px; height: 30px;
        background: rgba(0,255,255,0.6);
        border-radius: 50%;
      `
      this.joystickBase.appendChild(this.joystickKnob)

      this.nitroBtn = document.createElement('div')
      this.nitroBtn.style.cssText = `
        position: fixed; bottom: 80px; right: 60px;
        width: 80px; height: 80px;
        background: rgba(0,136,255,0.3);
        border: 2px solid rgba(0,255,255,0.6);
        border-radius: 50%;
        z-index: 500; display: flex;
        align-items: center; justify-content: center;
        color: #00ffff; font-family: 'Courier New', monospace;
        font-size: 0.85em; letter-spacing: 1px;
        touch-action: none;
      `
      this.nitroBtn.textContent = 'NITRO'

      document.body.appendChild(this.joystickBase)
      document.body.appendChild(this.nitroBtn)

      this.joystickBase.addEventListener('touchstart', (e) => {
        this.active = true
        this.startX = e.touches[0].clientX
        this.startY = e.touches[0].clientY
      })
      this.joystickBase.addEventListener('touchmove', (e) => {
        if (!this.active) return
        this.dx = e.touches[0].clientX - this.startX
        this.dy = e.touches[0].clientY - this.startY
        const maxD = 35
        const clampedX = Math.max(-maxD, Math.min(maxD, this.dx))
        const clampedY = Math.max(-maxD, Math.min(maxD, this.dy))
        this.joystickKnob.style.transform = `translate(${clampedX}px, ${clampedY}px)`
        this.forward  = this.dy < -10
        this.backward = this.dy > 10
        this.left     = this.dx < -10
        this.right    = this.dx > 10
      })
      this.joystickBase.addEventListener('touchend', () => {
        this.active = false
        this.dx = this.dy = 0
        this.joystickKnob.style.transform = ''
        this.forward = this.backward = this.left = this.right = false
      })
      this.nitroBtn.addEventListener('touchstart', () => { this.nitro = true })
      this.nitroBtn.addEventListener('touchend',   () => { this.nitro = false })
    }
  }

--- 6-1. 빌드 전 최종 점검 ---

아래 사항을 확인하고 문제가 있으면 수정한다:

  a) vite.config.ts의 base가 '/neon-runner/' 인지 확인

  c) TypeScript 컴파일 오류 없는지 확인: npx tsc --noEmit
  d) 개발 서버에서 정상 작동하는지 확인: npm run dev

--- 6-2. GitHub 원격 저장소 설정 ---

아래 명령을 실행한다 (YOUR_GITHUB_ID와 neon-runner는 실제 값으로 대체):

  git remote -v

remote가 없으면:
  git remote add origin https://github.com/YOUR_GITHUB_ID/neon-runner.git

--- 6-3. 메인 브랜치 푸시 ---

  git branch -M main
  git push -u origin main

--- 6-4. 프로덕션 빌드 ---

  npm run build

빌드 오류 발생 시:
  - TypeScript 타입 오류: 해당 파일 수정 후 재빌드
  - 모듈 not found: npm install 재실행 후 재빌드
  - 최대 3회 자동 재시도

--- 6-5. GitHub Pages 배포 ---

  npm run deploy

이 명령은 내부적으로 실행된다:
  1. npm run build (dist 폴더 생성)
  2. gh-pages -d dist (gh-pages 브랜치에 dist 내용 push)

GitHub 저장소 Settings → Pages → Source가 'gh-pages' 브랜치로 설정되어 있어야 한다.
